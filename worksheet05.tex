\begin{enumerate}
\item \textbf{Properties of exponent} 
Use the properties to simplify the following terms. \textbf{Do not} blindly apply the formulas.
\begin{tasks}[after-item-skip = 30pt, after-skip = 30pt](2)
\task $2^3 \times 2^4$
\task $3^5 \times 3^{-7}$
\task $2^8 \times 3^8$
\task $4^{-3} \times 5^{-3}$
\task $5^{10} \times 4^{5}$
\task $(2^4)^5$
\end{tasks}

\item \textbf{Exponential equation} 
Solve the following exponential equations
\begin{tasks}[after-item-skip = 30pt, after-skip = 30pt](2)
\task $4^x = \frac{1}{64}$
\task $9^{x+1} = 27$
\task $5^{x} = 25^{2x-3}$
\task $9^{x} - 3^{x+1} = 0$
\task $2^{x} + 2^{x+1} = 96$
\task $4^x - 12 (2^x) + 32 = 0$ 
\end{tasks}

\item \textbf{Applications of exponent} 
\begin{tasks}[after-item-skip = 100pt, after-skip = 100pt](1)
\task The fox population in a certain region has an annual growth rate of 9\% per year. In the year 2012, there were 23,900 fox counted in the area. What is the fox population predicted to be in the year 2020?
\task A car was valued at \$38,000 in the year 2007. By 2013, the value had depreciated to \$11,000. If the car’s value continues to drop by the same percentage, what will it be worth by 2017?
\task If we opended an investment account with an initial deposit of \$4,000 and with an annual interest rate of 7\%, compounding quarterly, how much will the account be worth in 9 years? 
\end{tasks}

\item \textbf{Properties of logarithm} 
Use the properties to simplify the following terms.
\begin{tasks}[after-item-skip = 40pt, after-skip = 40pt](2)
\task $\log_6 4 + \log_6 9$
\task $\log_2 20 - \log_2 5$
\task $\frac{\log_3 16}{\log_3 2}$
\task $\log_2 (5^{-8})$ 
\end{tasks}

\item \textbf{Logarithmic equation} 
Solve the following logarithmic equations. Denote $\log x = \log_{10} x$ and $\ln x = \log_{e} x$.
\begin{tasks}[after-item-skip = 50pt, after-skip = 50pt](2)
\task $\log_{2}(x+3) = 5$
\task $\log_{5}(2x-1) = 2$
\task $\log(x) + \log(x-3) = 1$
\task $\ln(x^2-1) = \ln(3x-1)$
\task $\log_{3}(x+4) - \log_{3}(x) = 2$
\task $2\log(x) = \log(5x-6)$
\end{tasks}

\item \textbf{Applications of logarithm} For the following problems, use these approximations
\[\log_{10} 2 \approx 0.301, \quad \log_{10} 3 \approx 0.477, \quad \log_5 10 \approx 1.4306\]
\begin{tasks}[after-item-skip = 100pt, after-skip = 100pt](1)
\task How many digits does $3^{100}$ have in base 10?
\task How many digits does $10^{100}$ have in base 5?
\task Which number is larger: $2^{100}$ or $3^{60}$? Prove by counting the number of digits for each number.
\end{tasks}

\end{enumerate}
