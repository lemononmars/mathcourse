\item \textbf{Linear equation} Solve the following linear equations.
\begin{tasks}[after-skip = 50pt, after-item-skip = 50pt](3)
\task $x + 3 = 1$
\task $x + 3 = 2x-1$
\task $-2x - 5 = 3x - 8$
\end{tasks}

\item \textbf{System of equations} Solve the following systems of equations.
\begin{tasks}[after-skip = 100pt](3)
\task
\begin{eqnarray*}
x + y &=& 6\\
2x - y &=& 3
\end{eqnarray*}
\task
\begin{eqnarray*}
2x + 3y &=& 7 \\
4x - 3y &=& 5
\end{eqnarray*}
\task
\begin{eqnarray*}
3x + 4y &=& 6 \\
2x - 5y &=& -1
\end{eqnarray*}
\end{tasks}

\item \textbf{Quadratic equation}
\begin{enumerate}
\item Find the roots following quadratic functions using the formula
\[ x =  \frac{-b \pm \sqrt{b^2 - 4ac}}{2a}.\]
\begin{tasks}[after-skip = 100pt](3)
\task $f(x) = x^2 +5x+6$
\task $f(x) = x^2 - x - 12$
\task $f(x) = 3x^2 +5x - 2$
\end{tasks}

\item Find the roots of the following quadratic functions by factoring.
\[(x-m_1)(x-m_2) = x^2 - (m_1+m_2)x + m_1m_2.\]
\begin{tasks}[after-skip = 100pt](3)
\task $f(x) = x^2 +3x+2$
\task $f(x) = x^2 + 7x - 18$
\task $f(x) = 3x^2 +5x - 2$
\end{tasks}
\end{enumerate}
