\item \textbf{Function evaluation}

\begin{enumerate}
\item Let $f(x) = 2x+3$. Find the followings.
\begin{tasks}[after-skip = 20pt](4)
\task $f(0)$
\task $f(2)$
\task $f(-3)$
\task $f(a+b)$
\end{tasks}
\item Let $g(x) = x^2+2x$. Find the followings.
\begin{tasks}[after-skip = 20pt](4)
\task $g(1)$
\task $g(-1)$
\task $g(x+1)$
\task $g\left(\frac{1}{\theta}\right)$
\end{tasks}
\item Let $h(x) = \frac{x+3}{x-2}$. Find the followings.
\begin{tasks}[after-skip = 20pt](4)
\task $h(-2)$
\task $h(3)$
\task $h(7)$
\task $h(a+2)$
\end{tasks}
\end{enumerate}

% พีชคณิตของฟังก์ชัน
\item \textbf{Algebra of functions} 

\begin{enumerate}
\item Let $f(x) = 3x$ and $g(x) = x+4$ Find the followings.
\begin{tasks}[after-skip = 40pt](3)
\task $2f(x)$
\task $f(x) + g(x)$
\task $f(x)g(x)$
\end{tasks}
\item Let $f(x) = x^2+2$ and $g(x) = x^3-1$. Find the followings.
\begin{tasks}[after-skip = 40pt](3)
\task $-f(x)$
\task $f(x)+g(x)$
\task $f(x)g(x)$
\end{tasks}
\item Let $f(x) = x^2-1$ and $g(x) = x-1$. Find the followings.
\begin{tasks}[after-skip = 40pt](3)
\task $xg(x)$
\task $f(x) + xg(x)$
\task $\frac{f(x)}{g(x)}$
\end{tasks}
\end{enumerate}

\item \textbf{Composite function} 

Given $f(x)$ and $g(x)$, find $(f \circ g)(x)$ and $(g \circ f)(x)$
\begin{tasks}(3)
\task $f(x) = 3x, g(x) = x^2$
\task $f(x) = x^3,  g(x) = 2x-1$
\task $f(x) = x^2+1, g(x) = x^2-1$
\end{tasks}

\item \textbf{Plotting Lines} Graph the following functions for $x$ in $[0,2]$.
\begin{tasks}[after-skip = 100pt, after-item-skip = 100pt](1)
\task $f(x) = \lfloor x \rfloor$ where $\lfloor x \rfloor$ is the largest integer that is smaller than or equals to $x$. (This is called the floor function.) For instance $f(1.1) = 1, f(2.9) = 2, f(-1.5)=-2$ and $f(5)=5$
\task $f(x) =$ the fractional part of $x$. For instance, $f(1.2) = 0.2, f(-2.01) = 0.01$ and $f(4) = 0$
\end{tasks}

\task $x^2 = x+2$
