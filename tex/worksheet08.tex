\begin{enumerate}
\item \textbf{Matrix Basics} 

Compute the following matrix operations.
\begin{tasks}[after-skip = 60pt, after-item-skip = 60pt](3)
\task $3 \begin{pmatrix} 1 & 2 \\ -1 & 0 \end{pmatrix}$
\task $\begin{pmatrix} 1 & 2 \\ 4 & 5 \end{pmatrix} + \begin{pmatrix} 0 & -1 \\ -3 & 1 \end{pmatrix}$
\task $\begin{pmatrix} 1 & -2 \\ -4 & 5 \end{pmatrix} - \begin{pmatrix} 3 & -2 \\ 1 & -4 \end{pmatrix}$
\task $2 \begin{pmatrix} 3 & 1 \\ -1 & 0 \end{pmatrix} - 3 \begin{pmatrix} 0 & 1 \\ 3 & -2 \end{pmatrix}$
\task $\begin{pmatrix} 1 & 0 \\ 2 & 3 \end{pmatrix} \begin{pmatrix} -1 \\ 4 \end{pmatrix}$
\task $\begin{pmatrix} 5 & -2 \\  1 & 3 \end{pmatrix} \begin{pmatrix} 0 \\ 2 \end{pmatrix}$
\end{tasks}


\item \textbf{Transformations Using Matrices} 

Transform the following vectors by identifying the transformation matrix and computing the multiplication.
\begin{tasks}[after-skip = 60pt, after-item-skip = 60pt](2)
\task Reflect $\begin{pmatrix} 2 \\ 6 \end{pmatrix}$ across the $x$-axis.
\task Scale $\begin{pmatrix} 3 \\ -2 \end{pmatrix}$ by a factor of 4.
\task Rotate $\begin{pmatrix} 2 \\ -1 \end{pmatrix}$ clockwise by $180^\circ$.
\task Rotate $\begin{pmatrix} 4 \\ -5 \end{pmatrix}$ clockwise by $90^\circ$.
\task Rotate $\begin{pmatrix} -3 \\ 0 \end{pmatrix}$ counter-clockwise by $90^\circ$.
\task Rotate $\begin{pmatrix} 0 \\ 2 \end{pmatrix}$ counter-clockwise by $45^\circ$.
\end{tasks}

\newpage

\item \textbf{Matrix Multiplication} 

Compute the following matrix products.
\begin{tasks}[after-skip = 60pt, after-item-skip = 60pt](2)
\task $\begin{pmatrix} 2 & 1 \\ 0 & 2 \end{pmatrix} \begin{pmatrix} 1 & 3 \\ 4 & 0 \end{pmatrix}$
\task $\begin{pmatrix} -1 & 2 \\ -4 & 3 \end{pmatrix} \begin{pmatrix} -2 & 1 \\ 4 & 5 \end{pmatrix}$
\end{tasks}

\item \textbf{Composition of Transformations}

Describe the result of the following compositions, both verbally (in terms of geometric transformations) and mathematically (as a single matrix).
\begin{tasks}[after-skip = 60pt, after-item-skip = 60pt](1)
\task Dilation $\begin{pmatrix} 4 & 0 \\ 0 & 4 \end{pmatrix}$ and rotation by $90^{\circ} \quad \begin{pmatrix} 0 & -1\\ 1& 0 \end{pmatrix}$
\task Dilation $\begin{pmatrix} 2 & 0 \\ 0 & 2 \end{pmatrix}$ and rotation by $45^{\circ} \quad \begin{pmatrix} \frac{\sqrt{2}}{2} & -\frac{\sqrt{2}}{2} \\ \frac{\sqrt{2}}{2} & \frac{\sqrt{2}}{2} \end{pmatrix}$
\task Rotation by $90^{\circ} \quad \begin{pmatrix} 0 & -1 \\ 1 & 0 \end{pmatrix}$ and rotation by $90^{\circ} \quad \begin{pmatrix} 0 & -1\\ 1& 0 \end{pmatrix}$
\end{tasks}

\end{enumerate}
