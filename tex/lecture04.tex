\begin{frame}
\begin{defn}{}{}
If a variable $y$ depends on a variable $x$ in such a way that each value of $x$ determines exactly one value of $y$, we say that $y$ is a function of $x$.
\end{defn}

Five common methods for representing functions are:
\begin{enumerate}
\item Numerically by tables
\item Abstractly by sets
\item Algebraically by formulas
\item Geometrically by graphs
\item Verbally
\end{enumerate}
\end{frame}

\begin{frame}
\frametitle{1. Numerically by tables}
A function can be represented as an input-output lookup table (e.g., Character Level vs. Max HP):
\begin{table}[h!]
\centering
\begin{tabular}{|c||c|c|c|c|c|c|}
\hline
\textbf{Level ($x$)} & 1 & 2 & 3 & 4 & 5 & 6 \\
\hline
\textbf{Max HP ($y$)} & 100 & 160 & 240 & 340 & 460 & 600 \\
\hline
\end{tabular}
\end{table}
\medskip
For every input level $x$, there is exactly one determined output HP value $y$.
\end{frame}

\begin{frame}
\frametitle{5. Verbally}
Sometimes functions are described in words.
\begin{remark}{Newton's Law of Universal Gravitation}
The gravitational force of attraction between two bodies in the Universe is directly proportional to the product of their masses and inversely proportional to the square of the distance between them.
\end{remark}
This is the verbal description of the formula
\[F = G \frac{m_1 m_2}{r^2}\]
where $F$ is defined as a function of $r$.
\end{frame}


\begin{frame}
\frametitle{Formal definition}
\begin{defn}{}{}
A function needs to give only one output for any input.
\medskip

We write 
\[ f: A \rightarrow B\] 
to represent a function $f$ from the set $A$, called \textbf{Domain}, to the set $B$, called \textbf{Codomain} (where the actual outputs form the \textbf{Range} $f(A) \subseteq B$).
\end{defn}
\end{frame}

\begin{frame}{Functions vs Non-Functions in Game Design}
\begin{block}{Functions in Games (Deterministic: 1 Input $\rightarrow$ Exactly 1 Output)}
\begin{itemize}
\item $\text{XPRequirement}(\text{Level})$: Level $5 \rightarrow 1,500\text{ XP}$ (fixed threshold). 
\item $\text{Speed}(\text{Agility})$: Deterministic stat formula ($v = 10 + 0.5 \times \text{Agi}$). 
\item $\text{ProjectileY}(\text{Time } t)$: Parabolic flight path height at time $t$.
\end{itemize}
\end{block}

\begin{block}{Non-Functions in Games (Non-Deterministic: 1 Input $\rightarrow$ Multiple Outputs)}
\begin{itemize}
\item $\text{LootChest}(\text{Tier } 1) \rightarrow \{\text{Common Sword}, \text{Potion}, 50\text{ Gold}\}$ (RNG loot drop). 
\item $\text{RollDice}() \rightarrow \{1, 2, 3, 4, 5, 6\}$ (Random dice outcome). 
\item Branching dialogue choices: $[\text{Talk to NPC}] \rightarrow \{\text{Accept Quest}, \text{Refuse}\}$.
\end{itemize}
\end{block}
\end{frame}

\begin{frame}[t]
\frametitle{Function evaluation}
Given a function $f$ and a real number $a$ in the domain, we can define $f(a)$ to be the value when we substitute $x=a$ in the function. In other words, we replace all occurences of $x$ by $a$ 

\begin{ex}
Let $f(x) = 2x+1$. Compute the followings.
\begin{tasks}(4)
\task $f(0)$
\task $f(1)$
\task $f(0.5)$ 
\task $f(100)$
\end{tasks}
\end{ex}
\begin{sol}
\begin{eqnarray*} 
\mbox{When } x=0 &\mbox{ we have }&  f(0) = 2(0) + 1  = 1 \\ 
\mbox{When } x=1 &\mbox{ we have }&  f(1) = 2(1) + 1  = 2+1  = 3\\ 
\mbox{When } x=0.5 &\mbox{ we have }&  f(0.5) = 2(0.5) + 1  = 1 + 1  = 2\\ 
\mbox{When } x=100 &\mbox{ we have }&  f(100) = 2(100) + 1  = 200 + 1  = 201
\end{eqnarray*}
\end{sol}
\end{frame}


\begin{frame}[t]
\begin{ex}
Let $f(x) = x^2+x$. Compute the followings.
\begin{tasks}(4)
\task $f(0)$
\task $f(2)$
\task $f(-3)$ 
\task $f\left(\frac{1}{3}\right)$
\end{tasks}
\end{ex}
\begin{sol}
\begin{tasks}(1)
\task $f(0)  = 0^2 + 0 = 0$ 
\task $f(2)  = 2^2 + 2 = 4+2 = 6$ 
\task $f(-3)  = (-3)^2 + (-3)  = 9 - 3  = 6$ 
\task $f\left(\frac{1}{3}\right) = \left(\frac{1}{3}\right)^2 + \frac{1}{3} = \frac{1}{9} + \frac{1}{3} = \frac{4}{9}$
\end{tasks}
\end{sol}
\end{frame}

\begin{frame}[t]
\frametitle{Remarks}
\begin{remark}{}
We don't need to let $x$ be a number. It can be \textbf{anything}. For instance, let $f(x) = x^2+3x$. Then, we can write the followings.
\begin{slide}
\begin{eqnarray*}
f(y) &=& y^2+3y \\ 
f(\Xi) &=& \Xi^2 + 3\Xi \\ 
f(x+5) &=& (x+5)^2 + 3(x+5) \\ 
f(x+h) &=& (x+h)^2 + 3(x+h) \\ 
f\left(\frac{1}{x}\right) &=& \left(\frac{1}{x}\right)^2 + 3\left(\frac{1}{x}\right) 
\end{eqnarray*}
\end{slide}
\end{remark}
\end{frame}

\begin{frame}
\frametitle{Algebra of Functions}
Algebra of functions refers to operations on functions, specifically arithmetic operations.
\begin{prop}{}{}
Let $f$ and $g$ be functions and $c$ be a constant.
\begin{enumerate}
\item $(cf) (x) = c ( f(x) )$ 
\item $(f+g) (x) = f(x) + g(x)$ 
\item $(f-g) (x) = f(x) - g(x)$
\item $(f \cdot g) (x) = f(x) \cdot g(x) $
\item $ \left(\frac{f}{g}\right)(x) = \frac{f(x)}{g(x)}$ where $g(x) \neq 0$
\end{enumerate}
\end{prop}
\end{frame}


\begin{frame}[t]
\begin{ex}
Let $f(x) = x+1$ and $g(x) = 2x-3$. Find the following.
\begin{tasks}(4)
\task $(2f)(x)$
\task $(f+g)(x)$ 
\task $(f - g)(x)$
\task $(3f + 2g)(x)$
\end{tasks}
\end{ex}
\begin{sol}
\begin{eqnarray*}
(2f)(x)  &=& 2f(x)  \\ 
&=& 2(x+1)  = 2x+2 \\ 
(f+g)(x)  &=& (x+1) + (2x-3)  \\ 
&=& (x+2x) + (1-3)  = 3x-2 \\ 
(f-g)(x)  &=& (x+1) - (2x-3) \\ 
&=& (x+1) + (-2x + 3)  = (x-2x) + (1+3)  = -x+4 \\ 
(3f + 2g)(x) &=& 3f(x) + 2g(x) \\ 
&=& 3(x+1) + 2(2x-3)  = (3x+3) + (4x-6)  = 7x-3
\end{eqnarray*}
\end{sol}
\end{frame}


\begin{frame}[t]
\begin{ex}
Let $f(x) = x+1$ and $g(x) = 2x-3$. Find the followings.
\begin{tasks}(2)
\task $(fg)(x)$
\task $\left( \frac{f}{g} \right)(x)$
\end{tasks}
\end{ex}
\begin{sol}
\begin{eqnarray*}
(fg)(x)  &=& (x+1)(2x-3) \\ 
&=& 2x^2-3x+2x-3  = 2x^2-x-3\\ 
\left(\frac{f}{g} \right)(x)&=& \frac{x+1}{2x-3}
\end{eqnarray*}
We can leave the fraction as is.
\end{sol}
\end{frame}

\begin{frame}{Composite Functions: The Pipeline Concept}
A \textbf{composite function} is simply chaining two (or more) functions into a pipeline.
\begin{itemize}
\item The \textbf{output} of the first function becomes the \textbf{input} of the next function.
\item Think of it as an assembly line or sequential calculation in a game engine.
\end{itemize}

\begin{frame}{Composite Function: Notation and Rule}
\begin{defn}{Composite Function}{}
Given functions $f$ and $g$, the \textbf{composite function} $(f \circ g)$ is defined by:
\[ (f \circ g)(x) = f(g(x)) \]
\end{defn}

\textbf{How to read and evaluate the notation:}
\begin{itemize}
\item \textbf{Meaning of $\circ$}: Read $(f \circ g)$ as \textbf{``$f$ after $g$''} (apply $g$ first, then apply $f$). 
\item \textbf{Inside-Out Rule}: Always evaluate the \textbf{inner function} $g(x)$ first, then feed the result into the \textbf{outer function} $f$. 
\item \textbf{Order Matters}: In general, $(f \circ g)(x) \neq (g \circ f)(x)$!
\end{itemize}
\end{frame}

\begin{frame}{Example in Game Design: Damage Pipeline}
\begin{block}{Sequential Combat Calculations}
\begin{itemize}
\item \textbf{Step 1 (Base Damage from Level):}
\[ g(\text{lvl}) = 15 \times \text{lvl} + 20 \]
\item \textbf{Step 2 (Critical Hit Buff - Double Damage):}
\[ f(\text{dmg}) = 2 \times \text{dmg} \]
\end{itemize}
\end{block}
use
\medskip
The \textbf{composite final damage function} $(f \circ g)$ combines both steps:
\[ (f \circ g)(\text{lvl}) = f(g(\text{lvl})) = 2(15 \times \text{lvl} + 20) = 30 \times \text{lvl} + 40 \]
use
\begin{itemize}
\item For a Level 4 player:
\[ (f \circ g)(4) = 30(4) + 40 = 160\text{ damage} \]
\end{itemize}
\end{frame}

\begin{frame}[t]
\begin{ex}
Let $f(x) = x+1$ and $g(x) = x^2$. Find the followings.
\begin{tasks}(2)
\task $(f \circ g)(x)$
\task $(g \circ f)(x)$
\task $(f \circ f)(x)$
\task $(g \circ g)(x)$
\end{tasks}
\end{ex}
\begin{sol}
\begin{eqnarray*}
(f \circ g)(x)  &=& f(\alert{g(x)}) = \alert{g(x)} + 1 = x^2 + 1 \\ 
(g \circ f)(x)  &=& g(\green{f(x)})  = \green{f(x)}^2  = \green{(x+1)}^2
\end{eqnarray*}
Notice that $(f \circ g)(x) \neq (g \circ f)(x)$, so the order matters.
\begin{eqnarray*}
(f \circ f)(x)  &=& f(\green{f(x)})  = \green{f(x)} + 1  = \green{x+1} + 1  = x+2 \\ 
(g \circ g)(x)  &=& g(\red{g(x)})  = (\red{g(x)})^2  = (\red{x^2})^2  = x^4
\end{eqnarray*}
\end{sol}
\end{frame} 


\begin{frame}[t]
\begin{ex}
Let $f(x) = \sqrt{x} $ and $g(x) = 2x+1$. Find the followings.
\begin{tasks}(2)
\task $(f \circ g)(x)$
\task $(f \circ g)(4)$
\task $(g \circ f)(x)$
\task $(g \circ f)(4)$
\end{tasks}
\end{ex}
\begin{sol}
\begin{eqnarray*}
(f \circ g)(x)  &=& f(\alert{g(x)}) = \sqrt{\alert{g(x)}}  = \sqrt{2x+1} \\ 
(f \circ g)(4)  &=& \sqrt{2(4) + 1 }  = \sqrt{9}  = 3 \\ 
(g \circ f)(x)  &=& g(\green{f(x)})  = 2\green{f(x)}+1  = 2\green{\sqrt{x}} + 1 \\ 
(g \circ f)(4)  &=& 2\sqrt{4} + 1  =  2(2) + 1  = 5
\end{eqnarray*}
\end{sol}
\end{frame}

\begin{frame}
\frametitle{Graph Intersection}
You can use graph to \textbf{find} solutions to some problems. Suppose that we have graphs for $y = f(x)$ and $y = g(x)$. Let $(a,b)$ be an intersection of the graphs. Then we know that
\[ b = f(a) \text{ and } b = g(a) \]
Thus, we have $f(a) = g(a)$. This implies that $x=a$ is the solution to
\begin{block}{}
\[f(x) = g(x)\]
\end{block}
This works if the solutions are simple decimals, or if you only want an approximation. You still need to solve the equation algebraically if you want an exact answer that is irrational, like $x = \sqrt{3}-1$.
\end{frame}



\begin{frame}[t]
\begin{ex}
Use graphs to find the solutions of
\[ x+1 = 2x-4\]
\end{ex}
\begin{sol}
We plot the graphs of the function $f(x) = x+1$ and $g(x) = 2x-4$ and see that the intersection is at the point $(5,6)$. Thus, the solution is $x = 5$.
\addfig{0.6}{graph_intersection_ex1}{The intersection of $f(x) = x+1$ and $g(x) = 2x-4$.}{}
\end{sol}
\end{frame}


\begin{frame}[t]
\begin{ex}
Use graphs to find the solutions of
\[ x^2 = x\]
\end{ex}

\begin{sol}
We plot the graphs of the function $f(x) = x^2$ and $g(x) = x$ and see that the intersections are at the points $(0,0)$ and $(1,1)$. Thus, the solutions are $x = 0$ and $x=1$.
\addfig{0.5}{graph_intersection_ex2}{The intersections of $f(x) = x^2$ and $g(x) = x$.}{}
\end{sol}
\end{frame}

\begin{frame}
\frametitle{Graph Comparison}
Similarly, we can compare the graph of $y = f(x)$ and $y = g(x)$ to solve an inequality of the form
\[ f(x) \geq g(x)\]
The solutions will be intervals with intersections as the left end or the right end of the intervals.
\end{frame}


\begin{frame}[t]
\begin{ex}
Use graphs to find the solutions of
\[ x \geq x^2\]
\end{ex}
\begin{sol}
We plot the graphs of the function $f(x) = x$ and $g(x) = x^2$ and see that the part where $f(x) \geq g(x)$ is when $0 \leq x \leq 1$. Thus, the solutions are in the closed interval $[0, 1]$.
\addfig{0.5}{graph_ineq_ex1}{The region where $f(x) \geq g(x)$.}{}
\end{sol}
\end{frame}


\begin{frame}[t]
\begin{ex}
Use graphs to find the solutions of
\[ x^2-3x+3 \geq 1\]
\end{ex}
\begin{sol}
We plot the graphs of the function $f(x) = x^2-3x+3$ and $g(x) = 1$ (which is a constant function) and see that the region where $f(x) \geq g(x)$ is when $x \leq 1$ or $x \geq 2$. Thus, the solutions are in the union of intervals $(-\infty, 1] \cup [2, \infty)$.
\addfig{0.5}{graph_ineq_ex2}{The region where $f(x) \geq g(x)$.}{}
\end{sol}
\end{frame}

\begin{frame}{Polar Coordinates $(r, \theta)$}
In a \textbf{polar coordinate system}, each point $P$ is determined by:
\begin{itemize}
\item $r$: Radial distance from the origin (the \textbf{Pole})
\item $\theta$: Angle measured from the positive x-axis (\textbf{Polar Axis})
\end{itemize}

\begin{figure}[h!]
\centering
\begin{tikzpicture}[scale = 0.45]

\foreach \r in {1, 2, 3, 4} {
  \draw[gray!35, thick] (0,0) circle (\r cm);
  \node[gray!70, scale=0.55, below right] at (\r,0) {\r};
}
% Radial spokes
\foreach \a in {0, 30, 45, 60, 90, 120, 135, 150, 180, 210, 225, 240, 270, 300, 315, 330} {
  \draw[gray!25] (0,0) -- (\a:4.3cm);
  \node[gray!80, scale=0.5] at (\a:4.6cm) {$\a^\circ$};
}
% Polar axis
\draw[->, very thick] (0,0) -- (4.4,0) node[right, scale=0.7] {Polar Axis};
\draw[->, thick, gray] (0,0) -- (0,4.4);

% Example Point P(3, 45°)
\draw[thick, red, ->] (0,0) -- (45:3cm) node[midway, above left, scale=0.7] {$r=3$};
