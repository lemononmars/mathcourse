\begin{enumerate}
\item You roll a single die until you see a 5 or 6. Find the expected number of rolls.
\vspace{40pt}

\item (Exploding dice) You roll a single die. If you get a 6, you reroll the die and get an extra die to roll. These new dice could also generate more dice under the same condition! (There is a non-zero chance that you will use all the dice on earth... ). If you get 1--5, you stop rolling that die. Find the expected number of dice.
\vspace{40pt}

\item (Istanbul: Tea House) Announce a number between 3 and 12. Then roll two dice. If you roll equal to or greater than the announced number, take the announced number of coins from the general supply. Otherwise, only take 2 coins.

\begin{itemize}
\item Find the expected number of coins gained if you announce 5.
\item Find the expected number of coins gained if you announce 10.
\item Guess (or compute) the optimal announced number that yields the highest expected payoff.
\end{itemize}
\vspace{40pt}

\item (Istanbul: Red Mosque Tile) There is a special power tile that gives you one of the following options: After rolling two dice, you may:
\begin{itemize} 
\item turn one die to 4, or
\item reroll both dice once.
\end{itemize}
Determine when to choose the first and the second option.

\item Add or subtract the following complex numbers.
\begin{tasks}[after-skip=20pt, after-item-skip=15pt](4)
\task $(3 + 4i) + (2 - 5i)$
\task $(-1 + 3i) + (5 + 2i)$
\task $(5 + 2i) - (3 + 4i)$
\task $(2 - 3i) - (-4 + i)$
\end{tasks}


\item Find the product.
\begin{tasks}[after-skip=25pt](3)
\task $(2 + i)(3 - 4i)$
\task $(1 - 2i)(-3 + 5i)$
\task $(4 + 3i)(4 - 3i)$
\end{tasks}

\item Find the conjugate of each complex number.
\begin{tasks}[after-skip=20pt](3)
\task $z = 5 - 7i$
\task $z = -2 + 3i$
\task $z = 1 + 0.5i$
\end{tasks}

\item Simplify each quotient.
\begin{tasks}[after-skip=25pt](3)
\task $\dfrac{3 + 4i}{1 - 2i}$
\task $\dfrac{5 - i}{2 + 3i}$
\task $\dfrac{2 + i}{2 - i}$
\end{tasks}


\item Express each complex number in cis form $r\operatorname{cis}\theta$, where $\operatorname{cis}\theta=\cos\theta+i\sin\theta$. Use angles in degrees.
\begin{tasks}[after-skip=25pt](3)
\task $z = 1 + i$
\task $z = -\sqrt{3} + i$
\task $z = -\frac{1}{\sqrt{2}} - \frac{1}{\sqrt{2}}i$
\end{tasks}


\item Compute both $z_1z_2$ and $z_1/z_2$ using cis form, with angles in degrees.
\begin{tasks}[after-skip=25pt](2)
\task $z_1 = 2\operatorname{cis}(30^\circ),\; z_2 = 3\operatorname{cis}(60^\circ)$
\task $z_1 = 4\operatorname{cis}(45^\circ),\; z_2 = 2\operatorname{cis}(90^\circ)$
\end{tasks}

\item Use De Moivre's theorem to simplify. Give answers in cis form with angles in degrees.
\begin{tasks}[after-skip=20pt](3)
\task $\bigl(\operatorname{cis}(30^\circ)\bigr)^4$
\task $\bigl(2\operatorname{cis}(45^\circ)\bigr)^3$
\task $\bigl(3\operatorname{cis}(60^\circ)\bigr)^2$
\end{tasks}

\item Find all roots in cis form, with angles in degrees.
\begin{tasks}[after-skip=25pt](3)
\task Cube roots of $27$
\task Square roots of $-1$
\task Fourth roots of $16i$
\end{tasks}
 
\end{enumerate}
