\begin{enumerate}
\item \textbf{Function evaluation}

\begin{enumerate}
\item Let $f(x) = 2x+3$. Find the followings.
\begin{tasks}[after-skip = 25pt](3)
\task $f(0)$
\task $f(2)$
\task $f(-3)$
\end{tasks}
\item Let $g(x) = x^2+2x$. Find the followings.
\begin{tasks}[after-skip = 25pt](3)
\task $g(1)$
\task $g(-1)$
\task $g(x+1)$
\end{tasks}
\end{enumerate}


\item \textbf{Algebra of functions} 

\begin{enumerate}
\item Let $f(x) = 3x$ and $g(x) = x+4$. Find the followings.
\begin{tasks}[after-skip = 35pt](3)
\task $(2f)(x)$
\task $(f + g)(x)$
\task $(fg)(x)$
\end{tasks}
\item Let $f(x) = x^2+2$ and $g(x) = x^3-1$. Find the followings.
\begin{tasks}[after-skip = 35pt](3)
\task $-f(x)$
\task $(f+g)(x)$
\task $(fg)(x)$
\end{tasks}
\item Let $f(x) = x^2-1$ and $g(x) = x-1$. Find the followings.
\begin{tasks}[after-skip = 35pt](3)
\task $x \cdot g(x)$
\task $f(x) + x \cdot g(x)$
\task $\left(\frac{f}{g}\right)(x)$
\end{tasks}
\end{enumerate}

\item \textbf{Composite functions} 

Given $f(x)$ and $g(x)$, find the composite functions $(f \circ g)(x)$ and $(g \circ f)(x)$.
\begin{tasks}[after-skip = 45pt](3)
\task $f(x) = 3x, \quad g(x) = x^2$
\task $f(x) = x^3, \quad g(x) = 2x-1$
\task $f(x) = x^2+1, \quad g(x) = x^2-1$
\end{tasks}

\newpage
\item \textbf{Plotting Points and Graphs} Plot the points $(x,f(x))$ for the following functions with $x=-2,-1,0,1,2$ and connect the points to predict their shape.
\begin{tasks}[after-item-skip = 110pt, after-skip = 110pt](3)
\task $f(x) = \frac{x+1}{x+3}$
\task $f(x) = |2x+2|$
\task $f(x) = ||x|-1|$
\end{tasks}

\item \textbf{Special and Piecewise Functions} Graph the following functions for $x \in [0,2]$.
\begin{tasks}[after-skip = 75pt, after-item-skip = 75pt](1)
\task $f(x) = \lfloor x \rfloor$ where $\lfloor x \rfloor$ is the largest integer that is smaller than or equal to $x$ (the \textbf{floor function}). For instance, $f(1.1) = 1, f(2.9) = 2, f(-1.5)=-2$, and $f(5)=5$.
\task $f(x) = \{x\} = x - \lfloor x \rfloor$ (the \textbf{fractional part function}). For instance, $f(1.2) = 0.2, f(3.75) = 0.75, f(-1.2) = 0.8$, and $f(4) = 0$.
\end{tasks}

\item \textbf{Finding Solutions via Graphing} Use desmos.com (or your favorite graphing tool) to find all solutions of the following equations or inequalities.
\begin{tasks}[after-skip = 40pt](4)
\task $3x+1 = 2x+4$
\task $x^2 = x+2$
\task $x = \frac{1}{x}$
\task $x^2-3x+2 \leq 0$
\end{tasks}

\end{enumerate}
