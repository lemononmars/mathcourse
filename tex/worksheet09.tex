\begin{enumerate}
\item \textbf{Rule of product} 

Use rule of product to count the followings.
\begin{tasks}[after-skip = 40pt, after-item-skip = 40pt](2)
\task Select a smartphone from 6 different models and a sim from 3 different mobile companies.
\task Create a 5-digit number where all digits are odd numbers.
\task Take a photo for a family of five where they all stand in a single row.
\task Make a 6-digit lotterry ticket where each pair of consecutive numbers are different. (e.g. 012345 works but not 112345).
\end{tasks}


\item \textbf{Rule of sum} 

Use rule of sum to count the followings.
\begin{tasks}[after-skip = 30pt, after-item-skip = 30pt](1)
\task Choose one student to do an exercise in front of the room from 10 male students and 22 female students.
\task Choose a number from 1 to 100 (inclusive) which is divisible by 10, or has two repeating digits.
\end{tasks}

\item \textbf{Rule of product and sum} 

Use rule of product to product and rule of sum to count the followings.
\begin{tasks}[after-skip = 80pt, after-item-skip = 80pt](1)
\task Choose a 4-digit number where no two odd digits are adjacent, and no two even digits are adjacent.
\task Rearrange 3 letters from ABCDE (e.g. DAC, AEB) so that B and C do not both appear.
\task Create a number using digits 4, 5, 6, and 7 where all digits are different.
\end{tasks}

\newpage
\item \textbf{Factorial}  Simplify the following expressions as much as you can without using a calculator.
Use rule of sum to count the followings.
\begin{tasks}[after-skip = 50pt, after-item-skip = 50pt](4)
\task $(2+3)!$
\task $\frac{9!}{7!}$
\task $\frac{2!}{5!}$
\task $6! + 7!$
\task $12! - 11!$
\task $18 \times 17 \times 16 \times 15$
\task* $11 \times 9 \times 7 \times 5 \times 3 \times 1$ (challenging!)
\end{tasks}

\item \textbf{Permutation} Count the followings with permutation. 
\begin{tasks}[after-skip = 50pt, after-item-skip = 50pt](2)
\task Create a 3-digit number from digits 1,2,3,4,5 and 6 with no duplicates.
\task Arrange 5 letters from Thai alphabets (44 letters).
\task Create a 2x2 matrix where each entry is a different single-digit number (0-9).
\task Arrange 10 kids and 12 adults in a single row so that all kids are in front of adults.
\end{tasks}

\item \textbf{Combination}  Count the followings with combination.
\begin{tasks}[after-skip = 50pt, after-item-skip = 50pt](2)
\task Select 3 out of 5 courses.
\task Buy 6 donuts, each with different flavor, from 15 flavors.
\task Mix 4 colors from 9 colors.
\task Choose 4 letters from ``UNCOPYRIGHTABLE''.
\end{tasks}

\item \textbf{Mix}
Count the followings 
\begin{tasks}[after-skip = 50pt, after-item-skip = 50pt](1)
\task Form a debate team of 5 students from 4 seniors and 6 juniors such that there are more juniors than seniors in the team.
\task Create a 4-digit number containing exactly 2 different digits (e.g. 1112, 3030, 9888)
\end{tasks}

\end{enumerate}
